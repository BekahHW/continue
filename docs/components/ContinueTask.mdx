# Interactive Task Component Specification

## Overview

The `ContinueTask` component allows users to run tasks directly from documentation into their Continue extension. This creates an interactive learning experience where users can immediately try examples.

---

## Component API

### TypeScript Interface

```typescript
interface ContinueTaskProps {
  // Required
  title: string;                    // Task display name
  prompt: string;                   // The actual prompt to send to Continue
  mode: 'chat' | 'agent' | 'plan' | 'edit'; // Continue mode to use

  // Optional
  description?: string;              // Brief description of what task does
  category?: string;                 // Category for filtering (debugging, testing, etc.)
  estimatedTime?: string;            // "2-5 minutes"
  difficulty?: 'beginner' | 'intermediate' | 'advanced';
  prerequisites?: string[];          // What user needs before running
  expectedOutcome?: string;          // What will happen after running
  tags?: string[];                   // For search/filtering
  contextProviders?: string[];       // Suggested context (e.g., ['@CurrentFile', '@Terminal'])
}
```

## Component Behavior

### When Continue is Connected

1. **Detect Connection:** Component checks if Continue extension is active
2. **Show "Run in Continue" button:** Primary CTA
3. **On Click:**
   - Open Continue panel in user's IDE
   - Switch to specified mode (chat/agent/plan/edit)
   - Insert prompt into input field
   - Optionally auto-submit or let user review first

### When Continue is NOT Connected

1. **Show "Copy Prompt" button:** Secondary CTA
2. **Display Install CTA:** Link to installation page
3. **On Copy Click:**
   - Copy prompt to clipboard
   - Show success toast: "Prompt copied! Install Continue to run it directly."

## Implementation Details

### Continue Detection

```typescript
// Check if Continue is installed and active
async function isContinueConnected(): Promise<boolean> {
  try {
    // Method 1: Check for Continue API
    if (window.continueAPI) {
      return await window.continueAPI.isActive();
    }

    // Method 2: Check for VS Code extension
    if (window.vscode) {
      const extensions = await window.vscode.extensions.getAll();
      return extensions.some(ext => ext.id === 'Continue.continue');
    }

    // Method 3: Local storage flag (set by extension)
    return localStorage.getItem('continue-connected') === 'true';
  } catch {
    return false;
  }
}
```

### Run Task Function

```typescript
async function runTask(task: ContinueTaskProps) {
  try {
    // Open Continue panel
    await window.continueAPI.open();

    // Switch to specified mode
    await window.continueAPI.setMode(task.mode);

    // Insert prompt
    await window.continueAPI.setPrompt(task.prompt);

    // Track usage
    trackEvent('task_run', {
      title: task.title,
      category: task.category,
      mode: task.mode,
    });

    // Optional: Auto-submit or let user review
    if (task.autoSubmit) {
      await window.continueAPI.submit();
    }
  } catch (error) {
    console.error('Failed to run task:', error);
    // Fallback to copy
    copyToClipboard(task.prompt);
    showToast('Prompt copied to clipboard');
  }
}
```

### Copy to Clipboard

```typescript
function copyToClipboard(text: string) {
  navigator.clipboard.writeText(text).then(() => {
    showToast('✅ Prompt copied to clipboard!');
  });
}
```

## React Component Implementation

```tsx
import React, { useState, useEffect } from 'react';

interface ContinueTaskProps {
  title: string;
  prompt: string;
  mode: 'chat' | 'agent' | 'plan' | 'edit';
  description?: string;
  category?: string;
  estimatedTime?: string;
  difficulty?: 'beginner' | 'intermediate' | 'advanced';
  prerequisites?: string[];
  expectedOutcome?: string;
  tags?: string[];
  contextProviders?: string[];
}

export function ContinueTask(props: ContinueTaskProps) {
  const [isConnected, setIsConnected] = useState(false);
  const [isExpanded, setIsExpanded] = useState(false);
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    checkConnection();
  }, []);

  async function checkConnection() {
    const connected = await isContinueConnected();
    setIsConnected(connected);
  }

  async function runTask() {
    if (isConnected) {
      await window.continueAPI.open();
      await window.continueAPI.setMode(props.mode);
      await window.continueAPI.setPrompt(props.prompt);

      trackEvent('task_run', {
        title: props.title,
        category: props.category,
        mode: props.mode,
      });
    }
  }

  function copyPrompt() {
    navigator.clipboard.writeText(props.prompt);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);

    trackEvent('task_copied', {
      title: props.title,
      category: props.category,
      continueInstalled: isConnected,
    });
  }

  const difficultyStars = {
    beginner: '⭐',
    intermediate: '⭐⭐',
    advanced: '⭐⭐⭐',
  };

  return (
    <div className="continue-task">
      <div className="task-header">
        <h3>🤖 {props.title}</h3>
        <span className="task-mode">{props.mode}</span>
      </div>

      {props.description && (
        <p className="task-description">{props.description}</p>
      )}

      <div className="task-prompt">
        <code>{props.prompt}</code>
      </div>

      <div className="task-meta">
        {props.estimatedTime && (
          <span>⏱️ {props.estimatedTime}</span>
        )}
        {props.difficulty && (
          <span>{difficultyStars[props.difficulty]} {props.difficulty}</span>
        )}
        {props.category && (
          <span>🏷️ {props.category}</span>
        )}
      </div>

      {isExpanded && (
        <div className="task-details">
          {props.prerequisites && (
            <div>
              <h4>✅ Prerequisites</h4>
              <ul>
                {props.prerequisites.map((req, i) => (
                  <li key={i}>{req}</li>
                ))}
              </ul>
            </div>
          )}

          {props.expectedOutcome && (
            <div>
              <h4>🎯 Expected Outcome</h4>
              <p>{props.expectedOutcome}</p>
            </div>
          )}

          {props.contextProviders && (
            <div>
              <h4>💡 Suggested Context</h4>
              <p>{props.contextProviders.join(', ')}</p>
            </div>
          )}
        </div>
      )}

      <div className="task-actions">
        {isConnected ? (
          <button
            onClick={runTask}
            className="btn-primary"
            aria-label={`Run ${props.title} in Continue`}
          >
            ▶️ Run in Continue
          </button>
        ) : (
          <a
            href="/getting-started/install"
            className="btn-secondary"
          >
            Install Continue
          </a>
        )}

        <button
          onClick={copyPrompt}
          className="btn-secondary"
          aria-label="Copy prompt to clipboard"
        >
          {copied ? '✅ Copied!' : '📋 Copy Prompt'}
        </button>

        <button
          onClick={() => setIsExpanded(!isExpanded)}
          className="btn-link"
        >
          {isExpanded ? 'Less' : 'More'} Details
        </button>
      </div>
    </div>
  );
}
```
